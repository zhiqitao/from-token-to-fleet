<section class="block alt" id="overview">
  <div class="wrap">
    <p class="kicker">Overview</p>
    <h2 class="h2">Introducing Ouro</h2>
    <div class="overview">
      <div class="prose">
        <p>Instead of leaving reasoning to chain-of-thought after training, Ouro builds it into <b>pre-training</b>: iterative computation in latent space, an <b>entropy-regularized objective</b> that learns how deep to think, and <b>7.7T tokens</b> of data.</p>
        <p>Its gains come from better <b>knowledge manipulation</b>, not more knowledge storage, and its latent reasoning is <b>more faithful</b> than explicit chain-of-thought.</p>
      </div>
      <div class="features">
        <div class="feat"><div class="ic"><svg><use href="#i-loop"/></svg></div><h4>Looped architecture</h4><p>One shared stack, applied R&nbsp;=&nbsp;4 times.</p></div>
        <div class="feat"><div class="ic"><svg><use href="#i-gauge"/></svg></div><h4>Learned depth allocation</h4><p>Easy tokens exit early; hard tokens loop longer.</p></div>
        <div class="feat"><div class="ic"><svg><use href="#i-brain"/></svg></div><h4>Knowledge manipulation</h4><p>Better at composing facts, not storing more.</p></div>
        <div class="feat"><div class="ic"><svg><use href="#i-eye"/></svg></div><h4>Faithful latent reasoning</h4><p>Latent steps track the final answer closely.</p></div>
      </div>
    </div>
  </div>
</section>

<section class="block alt" id="strategy">
  <div class="wrap">
    <p class="kicker">Strategy</p>
    <h2 class="h2">How a LoopLM decides how deep to think</h2>
    
    <ol class="steps3">
      <li><span class="n">01</span><h4>Train every depth</h4><p>Every loop gets an LM loss, weighted by a learned exit distribution with a <b>uniform-prior</b> entropy term.</p></li>
      <li><span class="n">02</span><h4>Teach the exit gate</h4><p>A second stage trains only the gate: <b>continue</b> while the next loop still lowers the loss.</p></li>
      <li><span class="n">03</span><h4>Exit with Q-exit</h4><p>Stop once the cumulative exit probability reaches <b>q</b>, a deploy-time compute knob.</p></li>
    </ol>
    <figure class="fig-wide">

Inference: the model exits once the cumulative exit probability passes a threshold." loading="lazy" data-full="./static/images/ouro_strategy.webp">

  // search over the full list
  var search = document.getElementById('paperSearch'), noRes = document.getElementById('noResults');
  if (search) search.addEventListener('input', function () {
    var q = search.value.trim().toLowerCase(), any = false;
    document.querySelectorAll('details.group').forEach(function (g) {
      var n = 0;
      g.querySelectorAll('li').forEach(function (li) {
        var hit = !q || li.textContent.toLowerCase().indexOf(q) !== -1;
        li.hidden = !hit; if (hit) n++;
      });
      g.hidden = n === 0; if (n) any = true;
      if (q) g.open = n > 0;
    });
    noRes.style.display = any ? 'none' : 'block';
  });

  // lightbox
  var lb = document.getElementById('lightbox'), lbImg = document.getElementById('lbImg'), lbCap = document.getElementById('lbCap');
  function openLb(src, cap, alt) {
    lbImg.src = src; lbImg.alt = alt || ''; lbCap.textContent = cap || '';
    if (lb.showModal) lb.showModal(); else window.open(src, '_blank');
  }
  document.querySelectorAll('button.fig').forEach(function (b) {
    b.addEventListener('click', function () {
      var img = b.querySelector('img');
      openLb(img.getAttribute('src'), b.getAttribute('data-cap'), img.alt);
    });
  });
  document.querySelectorAll('img[data-full]').forEach(function (im) {
    im.addEventListener('click', function () {
      var cap = im.closest('figure') && im.closest('figure').querySelector('figcaption');
      openLb(im.getAttribute('data-full'), cap ? cap.textContent : '', im.alt);
    });
  });

  // performance size toggle
  var perf = document.getElementById('perf');
  if (perf) perf.querySelectorAll('.seg button').forEach(function (b) {
    b.addEventListener('click', function () {
      perf.setAttribute('data-size', b.getAttribute('data-size'));
      perf.querySelectorAll('.seg button').forEach(function (x) { x.setAttribute('aria-pressed', x === b ? 'true' : 'false'); });
    });
  });
  document.getElementById('lbClose').addEventListener('click', function () { lb.close(); });
  lb.addEventListener('click', function (e) { if (e.target === lb) lb.close(); });

  // active nav link
  var links = document.querySelectorAll('.nav-links a');
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) {
        if (en.isIntersecting) {
          links.forEach(function (a) { a.classList.toggle('active', a.getAttribute('href') === '#' + en.target.id); });
        }
      });
    }, { rootMargin: '-45% 0px -50% 0px' });
    ['news', 'vllm', 'overview', 'performance', 'strategy', 'models', 'pipeline', 'follow-ups', 'citation'].forEach(function (id) {
      var s = document.getElementById(id); if (s) io.observe(s);
    });
  }
})();
